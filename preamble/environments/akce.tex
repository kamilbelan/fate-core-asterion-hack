\usepackage{enumitem}  % For customizing itemize/enumerate environments

%%% List with action icons as bullets: \begin{akce} \iconitem{<icon>}{<text>} ... \end{akce}
%% The icon is two text lines tall and its top is aligned with the top of the first line;
%% one-line items are padded to two lines, so an icon never runs into the next item.
\newlength{\akceiconsize}

\newenvironment{akce}
  {\setlength{\akceiconsize}{\dimexpr\baselineskip+\ht\strutbox+\dp\strutbox\relax}%
   \begin{itemize}[
      label={},
      align=left,
      labelwidth=\akceiconsize,
      labelsep=0.8em,
      leftmargin=\dimexpr\akceiconsize+0.8em\relax,
      topsep=0.6\baselineskip,
      itemsep=0.6\baselineskip plus 2pt,
      parsep=0.4\baselineskip,
    ]}
  {\end{itemize}}

% Icon + text item
\NewDocumentCommand{\iconitem}{m +m}{%
    \item[{\raisebox{\dimexpr\ht\strutbox-\akceiconsize\relax}[\ht\strutbox][0pt]{%
        \includegraphics[height=\akceiconsize]{#1}}}]%
    #2\par
    \ifnum\prevgraf<2 \vspace{\baselineskip}\fi
}
