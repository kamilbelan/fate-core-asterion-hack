%%% Boxes of the rulebook: pravidlo, předmět, povolání, realistické hraní
%% All share one look (coloured background, thick rule on the left, title run into the text)
%% and are made by `\NewAsterionBox{<env>}{<label>}{<background>}{<rule>}`;
%% use them as `\begin{<env>}[<name>] ... \end{<env>}`.
%%
%% Robustness:
%%   - the width follows the current line width, so a box inside a list or `fullwidth`
%%     never sticks out of the text column (into the wide margin),
%%   - extra `\emergencystretch`, since the box is narrower than the text
%%     and monospace `\dov{}` words do not hyphenate,
%%   - breakable across pages; every part of a split box is closed on its own (`enhanced jigsaw`),
%%     the part ends with "pokračuje na další straně" and the next one repeats the title.
\tcbuselibrary{hooks} % for `before upper app`
\tcbset{
    asterionbox/.style={
        todolike,
        enhanced jigsaw,
        width=\linewidth,
        before upper app={\emergencystretch=3em\relax},
        pad before break=2.4em,    % room for "pokračuje..." at the bottom of a part
        pad after break=2.4em,     % room for the repeated title at the top of the next part
    },
}

%% Marks of a box split across pages (#1 = colour, #2 = label, #3 = name or NoValue)
\NewDocumentCommand{\asterionboxcontinues}{m}{%
    \node[anchor=south east, inner sep=0pt, font=\footnotesize\sffamily\itshape, text=#1]
        at ([xshift=-0.9em, yshift=1.2ex]interior.south east) {pokračuje na další straně};%
}
\NewDocumentCommand{\asterionboxcontinued}{mmm}{%
    \node[anchor=north west, inner sep=0pt, font=\footnotesize\sffamily, text=#1]
        at ([xshift=0.9em, yshift=-1.2ex]interior.north west)
        {\textbf{#2}\IfValueT{#3}{ (#3)} -- pokračování};%
}

%% Each box type also gets a hook style `asterion <env>` (empty by default), used below to give
%% some types extra options, e.g. adding them to the lists printed after the TOC (see `toc.tex`).
\NewDocumentCommand{\NewAsterionBox}{mmmm}{%
    \tcbset{asterion #1/.style={}}%
    \NewTColorBox{#1}{o}{%
        asterionbox,
        colback=#3,
        colframe=#4,
        title={#2\IfValueT{##1}{ \textmd{(##1)}}.},
        list entry={\IfValueTF{##1}{##1}{#2}},% name of the box in a list (if it is added to one)
        bookmark={#2\IfValueT{##1}{: ##1}},% PDF outline entry under the current section (phone navigation)
        overlay first={\asterionboxcontinues{#4}},
        overlay middle={\asterionboxcontinued{#4}{#2}{##1}\asterionboxcontinues{#4}},
        overlay last={\asterionboxcontinued{#4}{#2}{##1}},
        asterion #1,
    }%
}

%%% Lists of boxes printed after the TOC (`\contentsandlists` in `preamble/layout/toc.tex`)
%%   pravidla    - all `pravidlo` boxes (+ rules written as plain text, see `\pravidlovprehledu`)
%%   prozatimni  - all `prozatimni` boxes
%%   zbrane      - `predmet` boxes of weapons and armour
%%   predmety    - all other `predmet` boxes
%% Which list a `predmet` box goes to is given by `\predmetlist` at the place of the box;
%% `predmety.tex` switches it to `zbrane` for the weapons and armour section.
\newcommand*{\predmetlist}{predmety}
%% `\asterionaddtolist{<list>}` (in the box's `phantom` code): link target at the box, then the
%%   entry (`list entry`); the list name is expanded right away, so a later switch of
%%   `\predmetlist` cannot affect boxes whose page is not shipped out yet
\makeatletter
\NewDocumentCommand{\asterionaddtolist}{m}{%
    \MakeLinkTarget*{asterionbox.\thetcolorboxnumber}%
    \edef\asterion@addtolist{\noexpand\tcb@addcontentsline{#1}{tcolorbox}}\asterion@addtolist
}
\makeatother
\tcbset{
    asterion pravidlo/.style={phantom={\asterionaddtolist{pravidla}}},
    asterion prozatimni/.style={phantom={\asterionaddtolist{prozatimni}}},
    asterion predmet/.style={phantom={\asterionaddtolist{\predmetlist}}, statblock},
    asterion specialita/.style={statblock},
}

%%% Stat blocks: in `predmet` and `specialita` boxes the list of fields ("Popis:", "Výdrž:", …)
%%   has no bullets, a hanging indent and tight spacing, so a box is compact and easy to scan;
%%   nested lists (e.g. under "Vlastnosti:") keep their bullets
\tcbset{
    statblock/.style={before upper app={%
        \setlist[itemize,1]{label={}, leftmargin=1em, itemindent=-1em, labelsep=0pt,
                            topsep=0.2\baselineskip, itemsep=0.1\baselineskip, parsep=0pt}%
        \setlist[itemize,2]{label=\textbullet, leftmargin=1.4em, itemindent=0pt, labelsep=0.5em,
                            topsep=0pt, itemsep=0pt, parsep=0pt}%
    }},
}
%% add a rule that is not in a `pravidlo` box (e.g. a whole subsection) to the list of rules,
%% linking to the current place: `\pravidlovprehledu{Fateův fraktál}`
\NewDocumentCommand{\pravidlovprehledu}{m}{\addcontentsline{pravidla}{tcolorbox}{#1}}
