\usepackage{fancyhdr}
%%% Chapter without number, but included in header and TOC
\NewDocumentCommand{\chapternotnumbered}{m}{
    \chapter*{#1}
    \markboth{#1}{#1}                   
    \addcontentsline{toc}{chapter}{#1}
}

%%% Chapter and section formatting (unchanged for sections)
\usepackage[newparttoc, clearempty]{titlesec}
\titleformat{\section}   {\MakeLinkTarget[section]{}\Large\sffamily\bfseries}{\thesection}   {0.8em}{}
\titleformat{\subsection}{\MakeLinkTarget[subsection]{}\large\sffamily\bfseries}{\thesubsection}{0.8em}{}
\titleformat{\paragraph}[runin]{\MakeLinkTarget[paragraph]{}\normalsize\sffamily\bfseries}{\theparagraph}{0em}{}

%%% Part title formatting (unchanged)
\titleformat{\part}[display]
{\thispagestyle{empty}\sffamily}
{\LARGE \partname~\Romanbar{\thepart}}
{0.2em}{\fontsize{30pt}{36pt}\selectfont\bfseries}

%%% Chapter title formatting - Fate Core style
%% Big colored number in a box on the left, bold title aligned
%% (colour `FateBlue` is defined in `preamble/general/colors.tex`, it is shared with links)

%% sizes: number box, gap between box and title, font sizes of number and title
\ifMOBILE
    \def\chapheadbox{1.1cm}\def\chapheadgap{0.45cm}
    \def\chapheadnumfont{\fontsize{21pt}{21pt}}\def\chapheadtitlefont{\fontsize{15pt}{17pt}}
\else
    \def\chapheadbox{1.5cm}\def\chapheadgap{0.7cm}
    \def\chapheadnumfont{\fontsize{28pt}{28pt}}\def\chapheadtitlefont{\fontsize{20pt}{22pt}}
\fi
\NewDocumentCommand{\chaphead}{s O{} m}{
    \vspace*{-20pt}% adjust vertical space before title
    \noindent
    \begin{tikzpicture}[baseline=(title.base)]
        % Box with chapter number
        \node[fill=FateBlue,text=white,rounded corners=3pt,
              minimum width=\chapheadbox,minimum height=\chapheadbox,
              align=center,font=\sffamily\bfseries\chapheadnumfont\selectfont] (num)
              at (0,0) {\IfBooleanTF{#1}{}{#2}};
        % Chapter title (wraps within the text width, without hyphenation)
        \node[anchor=west,xshift=\chapheadgap,font=\sffamily\bfseries\chapheadtitlefont\selectfont,text=FateBlue,
              text width=\linewidth-\chapheadbox-\chapheadgap-0.7em,align=left] (title)
              at (num.east) {\hyphenpenalty=10000\MakeUppercase{#3}\par};% `\par` while the penalty is still set
    \end{tikzpicture}
    \par\vspace{\ifMOBILE 14pt\else 25pt\fi}
}

\makeatother
